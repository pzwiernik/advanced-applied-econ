\section{Overview}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\huge \textcolor{DarkRed}{What can we learn from observational data alone?}}
\end{center}
\end{frame}

\begin{frame}{What changes today?}
In Lecture 3 we gave causal meaning to DAGs using factorization, d-separation, and interventions.
Today we ask a harder question:
\begin{center}
\Large \textcolor{DarkBlue}{How much of the causal direction can be recovered without intervening?}
\end{center}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Constraint-based discovery}
Use conditional independence tests to recover the skeleton and the compelled colliders.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Extra identifying structure}
Use assumptions such as non-Gaussianity or invariant mechanisms to go beyond Markov equivalence.
\end{block}
\end{column}
\end{columns}

\vspace{2mm}
\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
\textbf{Main question:}
Which arrow directions are visible from conditional independence alone, and which require extra assumptions?
\end{beamercolorbox}
\end{frame}

\begin{frame}{Roadmap for today}
\begin{enumerate}
    \item constraint-based discovery and the PC algorithm
    \item why Markov equivalence is the main obstacle
    \item why linear Gaussian SEMs do not identify direction
    \item LiNGAM and the role of non-Gaussianity
    \item more realistic shocks: heteroskedasticity and mean independence
    \item a brief link to additive noise models
\end{enumerate}

\vspace{2mm}
\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Conditional independence narrows down the graph; asymmetry assumptions identify direction.
\end{beamercolorbox}
\end{frame}

\section{Constraint-based discovery}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 1: Constraint-based discovery}}
\end{center}
\end{frame}

\begin{frame}{Why conditional independence alone stops at a CPDAG}
Constraint-based discovery uses the logic of Lecture 3:
\begin{itemize}
    \item d-separation implies conditional independence,
    \item missing edges correspond to some separating set,
    \item colliders can be oriented from separating sets.
\end{itemize}

\begin{block}{}
From observational data we may learn
\begin{itemize}
    \item the \textbf{skeleton},
    \item the \textbf{v-structures},
    \item and therefore a \textbf{CPDAG}.
\end{itemize}
\end{block}

\begin{alertblock}{What this does \emph{not} buy us}
Without extra assumptions, we usually cannot orient every edge.
\end{alertblock}
\end{frame}

\begin{frame}{High-level idea of the PC algorithm}
The PC algorithm is the canonical constraint-based discovery method.

\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{block}{Skeleton phase}
\begin{enumerate}
    \item Start from the complete undirected graph.
    \item Test marginal and conditional independences.
    \item Remove an edge whenever some separating set is found.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Orientation phase}
\begin{enumerate}
    \item Use separating sets to orient v-structures.
    \item Apply graphical rules to orient additional compelled edges.
\end{enumerate}
\end{block}
\end{column}
\end{columns}
\begin{alertblock}{Output}
A \imp{CPDAG}, not necessarily a unique DAG.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
PC is a discovery procedure based on conditional independence, not on interventions.
\end{beamercolorbox}
\end{frame}



\begin{frame}{The PC algorithm: skeleton phase}
\begin{block}{Input}
Data on $X_1,\dots,X_m$, a conditional independence test, and a significance level $\alpha$.
\end{block}

\begin{block}{Skeleton phase}
\begin{enumerate}
    \item Initialize with the complete undirected graph on $\{1,\dots,m\}$.
    \item For conditioning sets of size $0,1,2,\dots$:
    \begin{itemize}
        \item for each adjacent pair $(i,j)$,
        \item test $X_i \indep X_j \mid X_S$ for candidate sets $S$,
        \item if independence is not rejected, remove edge $i-j$ and store $S$.
    \end{itemize}
\end{enumerate}
\end{block}

\begin{alertblock}{Interpretation}
Search for a small set of variables that explains away the apparent association between $X_i$ and $X_j$.
\end{alertblock}
\end{frame}

\begin{frame}{A toy discovery example}
Suppose the true DAG is
\begin{center}
\begin{tikzpicture}[>=Latex,scale=0.7,
vertex/.style={circle,draw,very thick,minimum size=6mm,inner sep=0pt}]
\node[vertex] (1) at (0,0) {$1$};
\node[vertex] (2) at (2,1.2) {$2$};
\node[vertex] (3) at (2,-1.2) {$3$};
\node[vertex] (4) at (4,0) {$4$};
\node[vertex] (5) at (6,0) {$5$};
\draw[->,very thick] (1)--(2);
\draw[->,very thick] (2)--(4);
\draw[->,very thick] (3)--(4);
\draw[->,very thick] (4)--(5);
\end{tikzpicture}
\end{center}

\begin{block}{Skeleton phase intuition}
\begin{itemize}
    \item marginal tests may already remove $1-3$ and $2-3$,
    \item then conditioning on $2$ may remove $1-4$ and $1-5$,
    \item conditioning on $4$ may remove $2-5$ and $3-5$.
\end{itemize}
\end{block}

\begin{alertblock}{Message}
The skeleton is found by asking which pairs can be separated by some conditioning set.
\end{alertblock}
\end{frame}

\begin{frame}{Orientation phase: where colliders come from}
Suppose the skeleton contains

\begin{center}
\begin{tikzpicture}[>=Latex,scale=1.0,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (1) at (0,0) {$1$};
\node[vertex,fill=red!15] (2) at (2,0) {$2$};
\node[vertex] (3) at (4,0) {$3$};
\draw[very thick] (1)--(2)--(3);
\end{tikzpicture}
\end{center}

with no edge between $1$ and $3$.
If the separating set for $(1,3)$ \emph{does not} contain node $2$, then PC orients
\[
1\to 2 \leftarrow 3.
\]

\begin{alertblock}{Why?}
If $2$ were a non-collider, then $2$ would have to belong to every minimal separator between $1$ and $3$.
\end{alertblock}

\begin{block}{}
After orienting all such colliders, PC applies \imp{Meek's rules} to propagate additional arrow directions,
while avoiding new v-structures and directed cycles.
\end{block}

%\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
%Separating sets do not only remove edges; they also identify colliders.
%Meek's rules then propagate the remaining compelled directions.
%\end{beamercolorbox}
\end{frame}




\begin{frame}[fragile]{A first PC example in \texttt{R}}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
library(pcalg)
library(MASS)
set.seed(1)

Sigma <- matrix(c(1,.7,0,.49,0,
                  .7,1,0,.7,0,
                  0,0,1,.6,0,
                  .49,.7,.6,1,.7,
                  0,0,0,.7,1), 5, 5)
X <- mvrnorm(500, mu = rep(0, 5),
             Sigma = Sigma)
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
suffStat <- list(C = cor(X), n = nrow(X))
pc.fit <- pc(suffStat,
             indepTest = gaussCItest,
             alpha = 0.01,
             labels = paste0("X", 1:5),
             method = "stable")

pc.fit
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
For Gaussian data, the PC algorithm uses partial-correlation tests to recover a CPDAG.
\end{alertblock}
\end{frame}

\begin{frame}{Why economists should be cautious with PC}
Constraint-based discovery is elegant, but it rests on demanding assumptions.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Statistical issues}
\begin{itemize}
    \item many CI tests,
    \item finite-sample error propagation,
    \item high-dimensional conditioning sets,
    \item sensitivity to near-unfaithfulness.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Economic issues}
\begin{itemize}
    \item latent confounding is common,
    \item Gaussianity may be implausible,
    \item observational data may identify only part of the structure.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
PC is best viewed as a disciplined way to learn a CPDAG, not as a button that magically discovers the true causal graph.
\end{beamercolorbox}
\end{frame}

\section{Non-Gaussian identification}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 2: Why Gaussian linear models are not enough}}
\end{center}
\end{frame}

\begin{frame}{Why linear Gaussian SEMs do not identify direction}
Take two centered Gaussian variables $(X,Y)$.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Forward representation}
\[
Y = aX + \varepsilon_Y,
\qquad
\varepsilon_Y \indep X.
\]
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Backward representation}
\[
X = bY + \varepsilon_X,
\qquad
\varepsilon_X \indep Y.
\]
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Key point}
The same bivariate Gaussian distribution admits both directions as valid linear SEMs with independent noise.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Gaussianity is convenient for estimation, but it hides causal direction.
\end{beamercolorbox}
\end{frame}

\begin{frame}{What kind of extra assumptions can break the symmetry?}
To identify direction from observational data, we need an asymmetry that is invisible in the Gaussian model.

\begin{block}{Three common strategies}
\begin{itemize}
    \item \textbf{Constraint-based methods:} rely on conditional independence and stop at a CPDAG.
    \item \textbf{Non-Gaussian linear models:} LiNGAM uses non-Gaussianity of the shocks.
    \item \textbf{Nonlinear noise models:} additive noise models use functional asymmetry.
\end{itemize}
\end{block}

\begin{alertblock}{Today's focus}
We stay in the linear SEM world and ask what non-Gaussian shocks can reveal.
\end{alertblock}
\end{frame}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 3: LiNGAM and non-Gaussianity}}
\end{center}
\end{frame}

\begin{frame}{LiNGAM assumptions}
LiNGAM stands for \imp{Linear Non-Gaussian Acyclic Model}.

\begin{alertblock}{Model}
\[
X = BX + \varepsilon,
\]
where
\begin{itemize}
    \item $B$ is compatible with a DAG,
    \item the errors $\varepsilon_1,\dots,\varepsilon_p$ are mutually independent,
    \item the errors are non-Gaussian (except possibly one),
    \item there are no hidden common causes.
\end{itemize}
\end{alertblock}

\begin{block}{Message}
LiNGAM keeps linearity and acyclicity, but replaces Gaussian noise by non-Gaussian noise.
\end{block}
\end{frame}

\begin{frame}{Why non-Gaussianity helps}
Independence is highly restrictive under linear transformations.

\begin{block}{Kac / Darmois--Skitovich intuition}
If independent variables remain independent after a nontrivial linear mixing,
then the variables must be Gaussian.
\end{block}

\begin{alertblock}{Consequence for linear SEMs}
In a non-Gaussian linear SEM, the mixing matrix in
\[
X = (I-B)^{-1}\varepsilon
\]
is identifiable up to scaling and permutation.
With the acyclicity assumption, this is enough to recover the causal ordering
and then the edge coefficients.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Non-Gaussianity breaks the symmetry that makes Gaussian linear SEMs directionally ambiguous.
\end{beamercolorbox}
\end{frame}

\begin{frame}{Why non-Gaussianity helps}

\begin{frame}{DirectLiNGAM: the main idea}
DirectLiNGAM avoids full ICA and instead looks for an exogenous variable first.

\begin{block}{Key insight}
If $X_j$ has no parents, then regressing every other variable on $X_j$ leaves residuals that are independent of $X_j$.
\end{block}

\begin{enumerate}
    \item Find a variable that is most independent of its regression residuals.
    \item Treat it as exogenous and remove its effect from the remaining variables.
    \item Repeat on the residual system.
\end{enumerate}

\begin{alertblock}{Output}
A causal ordering, hence a fully directed DAG in the basic LiNGAM model.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{A simple LiNGAM simulation in \texttt{R}}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
set.seed(1)
n <- 2000

x1 <- rt(n, df = 5)
x2 <- 0.8 * x1 + rt(n, df = 5)
x3 <- -0.6 * x1 + 0.7 * x2 +
      rt(n, df = 5)

X <- cbind(x1, x2, x3)
colnames(X) <- c("x1", "x2", "x3")
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
# one possible implementation
library(bnlearn)
fit <- direct.lingam(as.data.frame(X))
fit

# older implementations also exist,
# e.g. pcalg::LINGAM(X)
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
Shocks are non-Gaussian, so direction can become identifiable even though the model is linear.
\end{alertblock}
\end{frame}

\begin{frame}{Why this is relevant for economics}
Economic shocks are often far from Gaussian.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Examples}
\begin{itemize}
    \item asset-return shocks are heavy-tailed,
    \item firm-level outcomes are skewed,
    \item macro shocks often have outliers and asymmetries.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Opportunity}
What is a nuisance for Gaussian methods can become useful for identification.
\end{block}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Non-Gaussianity is not only a robustness issue; it can also be an identification resource.
\end{beamercolorbox}
\end{frame}

\section{Beyond strict independence}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 4: Beyond strict independence of shocks}}
\end{center}
\end{frame}

\begin{frame}{Why independent shocks may be too strong in economics}
The basic LiNGAM assumption is that the structural shocks are mutually independent.

\begin{alertblock}{Why this may fail}
In economic data, shocks often share volatility, exposure, or background uncertainty.
\end{alertblock}

\begin{columns}[T]
\begin{column}{0.53\textwidth}
\begin{block}{Examples}
\begin{itemize}
    \item common market volatility,
    \item sectoral exposure to the same aggregate news,
    \item strategic interaction or equilibrium feedback,
    \item dependence created by omitted but low-frequency macro conditions.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.43\textwidth}
\begin{block}{Consequence}
The clean ICA-style separation behind LiNGAM may become fragile or fail outright.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{A toy example with common volatility}
Consider a linear SEM
\[
X_1 = \varepsilon_1,
\qquad
X_2 = aX_1 + \varepsilon_2,
\]
but let
\[
\varepsilon_1 = \sigma U_1,
\qquad
\varepsilon_2 = \sigma U_2,
\]
where $U_1,U_2$ are independent and non-Gaussian, while the random scale $\sigma$ is shared.

\begin{alertblock}{What happens?}
Conditional on $\sigma$, the shocks are independent; marginally they are dependent through common volatility.
\end{alertblock}

\begin{block}{Economic reading}
This is a crude model of common market stress: shocks are not driven by a common mean factor, but their magnitude co-moves across equations.
\end{block}
\end{frame}

\begin{frame}{A mean-independence perspective}
A weaker requirement than full independence is often more realistic.

\begin{block}{Mean independence}
Instead of requiring
\[
\varepsilon_i \indep \varepsilon_j \quad (i\neq j),
\]
we may only require suitable \imp{mean-independence} conditions.
For instance, a noise term may have mean zero given the relevant past or given suitable summary information.
\end{block}

\begin{alertblock}{Why this matters}
Mean independence allows dependence in higher moments, such as shared volatility, while still preserving enough asymmetry for identification or estimation.
\end{alertblock}
\end{frame}

\begin{frame}{Your extension: a more realistic economic shock model}
The motivation is simple:

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Classical LiNGAM}
Independent non-Gaussian shocks.
Elegant, but often too clean for real economic systems.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{More realistic alternative}
Allow some dependence between shocks, but keep a directional asymmetry through mean independence or related restrictions.
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Message}
The goal is to keep the identification spirit of LiNGAM while allowing richer and more realistic shock structures.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
For economics, the right question is often not “are the shocks fully independent?” but “what weaker asymmetry is still credible?”
\end{beamercolorbox}
\end{frame}

\begin{frame}[fragile]{A toy simulation with common volatility}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
set.seed(1)
n <- 2000

sigma <- abs(rnorm(n)) + 0.3
u1 <- rt(n, df = 5)
u2 <- rt(n, df = 5)

e1 <- sigma * u1
e2 <- sigma * u2
x1 <- e1
x2 <- 0.8 * x1 + e2
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
par(mfrow = c(1,2))
plot(x1, x2, pch = 16, cex = .4)
plot(abs(e1), abs(e2), pch = 16,
     cex = .4)
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
The second plot reveals common volatility in the shocks. This violates strict independence, even though the directional SEM remains meaningful.
\end{alertblock}
\end{frame}

\section{Additive noise models}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 5: A brief look beyond linearity}}
\end{center}
\end{frame}

\begin{frame}{Additive noise models}
A different route to identifiability is nonlinear asymmetry.

\begin{block}{Additive noise model}
\[
Y = f(X) + \varepsilon,
\qquad
\varepsilon \indep X.
\]
Under suitable regularity conditions, the reverse representation
\[
X = g(Y) + \widetilde\varepsilon,
\qquad
\widetilde\varepsilon \indep Y
\]
usually does not exist.
\end{block}

\begin{alertblock}{Comparison with LiNGAM}
LiNGAM uses linearity plus non-Gaussianity.
Additive noise models use nonlinearity plus independent noise.
\end{alertblock}
\end{frame}

\begin{frame}{Where this connects to causal ML}
The common theme is the search for an orientation in which the mechanism looks stable and simple.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{LiNGAM}
Look for a linear model with independent non-Gaussian shocks.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Additive noise / invariance ideas}
Look for a direction in which residuals are independent or mechanisms remain stable across environments.
\end{block}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Across many modern methods, causal direction is identified by some form of asymmetry.
\end{beamercolorbox}
\end{frame}

\section{Conclusion}

\begin{frame}{Take-away}
\begin{itemize}
    \item Constraint-based discovery uses conditional independence to recover a CPDAG.
    \item Markov equivalence limits what can be learned from conditional independence alone.
    \item Linear Gaussian SEMs do not identify direction in general.
    \item LiNGAM uses non-Gaussianity to restore identifiability.
    \item For economic data, strict independence of shocks may be unrealistic; weaker asymmetries such as mean independence can be more credible.
    \item Beyond linearity, additive noise models provide another route to directional identification.
\end{itemize}

\vspace{2mm}
\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
Conditional independence narrows down the graph.  
Asymmetry assumptions identify direction.
\end{beamercolorbox}
\end{frame}


\section{Graphical time series}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Extension: graphical ideas for time series}}
\end{center}
\end{frame}

\begin{frame}{From static SEMs to dynamic graphical models}
For a multivariate time series, a natural model is
\[
X_t = A_1X_{t-1}+\cdots + A_pX_{t-p}+\varepsilon_t.
\]

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Directed lagged structure}
Zeros in the matrices $A_\ell$ mean that some lagged variables do not directly predict others.
This is the dynamic analogue of missing directed edges.
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Contemporaneous structure}
If $
\varepsilon_t\sim (0,\Sigma_\varepsilon),
$
then zeros in
\[
K_\varepsilon=\Sigma_\varepsilon^{-1}
\]
encode conditional independence among shocks at time $t$.
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
A graphical time-series model separates
\begin{itemize}
    \item \textbf{lagged directional effects} from
    \item \textbf{instantaneous conditional dependence of shocks.}
\end{itemize}
\end{alertblock}
\end{frame}


\begin{frame}{Why economists may care about graphical time series}
These ideas are natural in macroeconomics and finance.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Lagged graph}
A sparse lagged graph can summarize who predicts whom:
\begin{itemize}
    \item sectoral spillovers,
    \item macro transmission channels,
    \item Granger-causal relations in large VARs.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Shock graph}
A sparse graph for $\varepsilon_t$ can summarize contemporaneous residual links:
\begin{itemize}
    \item common financial stress,
    \item instantaneous market co-movement,
    \item residual dependence after lagged dynamics are removed.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}

Dynamic graphical models ask: who predicts whom over time, and which shocks still interact contemporaneously?
\end{beamercolorbox}
\end{frame}

\begin{frame}[fragile]{A simple workflow in \texttt{R}}
A practical strategy is:
\begin{enumerate}
    \item estimate a sparse VAR for lagged structure,
    \item extract residuals,
    \item fit a Gaussian graph to the residuals.
\end{enumerate}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
library(BigVAR)

# X is a T x m matrix of series
mod <- constructModel(X,
                      p = 1,
                      struct = "Basic",
                      gran = c(10, 10),
                      verbose = FALSE)
fit <- cv.BigVAR(mod)
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
# residual graph
Ehat <- residuals(fit)
S <- cov(Ehat)

library(glasso)
out <- glasso(S, rho = 0.05)

library(qgraph)
qgraph(-cov2cor(out$wi),
       layout = "spring")
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
The sparse VAR encodes lagged directed effects.
The residual precision matrix estimates contemporaneous conditional dependence among shocks.
\end{alertblock}
\end{frame}
