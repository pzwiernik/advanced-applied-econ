\begin{frame}{Why talk about positive dependence?}
Most of the course has focused on sparsity, conditional independence, and latent structure.
But another kind of useful structure is the \imp{sign} of dependence.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Economic examples}
\begin{itemize}
    \item asset returns often comove positively,
    \item macro indicators may move together in expansions and recessions,
    \item symptoms of a syndrome are often all positively associated,
    \item a one-factor CAPM-style model gives positive comovement.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Statistical payoff}
Sign constraints can act as regularization:
\begin{itemize}
    \item existence of estimators improves,
    \item fitted graphs can become sparse,
    \item covariance estimation can be more stable.
\end{itemize}
\end{block}
\end{column}

\end{frame}

\section{Positive dependence}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 1: Positive dependence and $\mtp$}}
\end{center}
\end{frame}

\begin{frame}{Positive association}
A random vector $X=(X_1,\ldots,X_m)$ is \imp{positively associated} if
for any two non-decreasing functions $\phi,\psi$,
\[
\cov\{\phi(X),\psi(X)\}\geq 0.
\]

\begin{block}{Interpretation}
If one collection of coordinates is large, then other increasing functions of the data also tend to be large.
\end{block}

\begin{alertblock}{Examples}
\begin{itemize}
    \item positively correlated stock returns,
    \item symptoms of a common disease,
    \item variables driven by a common positive latent factor.
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Multivariate total positivity of order two}
A density $f$ is \imp{$\mtp$} if for all $x,y$,
\[
f(x)f(y)\leq f(x\wedge y)\,f(x\vee y),
\]
where $x\wedge y$ and $x\vee y$ denote componentwise minima and maxima.

\begin{block}{Why this matters}
$\mtp$ is a strong and tractable form of positive dependence.
It implies positive association and has powerful closure properties.
\end{block}

\begin{alertblock}{Basic properties}
If $X$ is $\mtp$, then:
\begin{itemize}
    \item every marginal distribution is $\mtp$,
    \item every conditional distribution is $\mtp$,
    \item and $X$ is positively associated.
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{No Yule--Simpson paradox for signs}
Positive dependence behaves differently from arbitrary dependence.

\begin{block}{A useful intuition}
With unrestricted models, marginal and conditional associations may change sign in messy ways.
With $\mtp$ models, signs are much more stable.
\end{block}

\begin{alertblock}{Why this is interesting}
This means that positive dependence is not only a descriptive feature.
It also simplifies the conditional independence and graphical structure.
\end{alertblock}
\end{frame}

\section{Special cases}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 2: Gaussian and binary special cases}}
\end{center}
\end{frame}

\begin{frame}{Two important special cases}
\begin{beamercolorbox}[wd=\textwidth,sep=8pt]{result}
\textbf{Gaussian model:}
\[
X\sim N(0,\Sigma),\qquad K=\Sigma^{-1}.
\]
Then
\[
X \text{ is } \mtp
\qquad\Longleftrightarrow\qquad
K_{ij}\leq 0 \text{ for all } i\neq j.
\]
\end{beamercolorbox}

\bigskip

\begin{beamercolorbox}[wd=\textwidth,sep=8pt]{result}
\textbf{Binary Ising model:}
\[
p(x)\propto \exp(h^\top x + x^\top J x/2).
\]
Then
\[
X \text{ is } \mtp
\qquad\Longleftrightarrow\qquad
J_{ij}\geq 0 \text{ for all } i\neq j.
\]
\end{beamercolorbox}
\end{frame}

\begin{frame}{Gaussian $\mtp$ means an M-matrix precision}
In the Gaussian case, $\mtp$ is equivalent to saying that the precision matrix is an \imp{M-matrix}:
\[
K_{ii}>0,
\qquad
K_{ij}\leq 0 \ \ (i\neq j).
\]

\begin{block}{Consequences}
This implies:
\begin{itemize}
    \item all correlations are nonnegative,
    \item all partial correlations are nonnegative,
    \item the graph is an \emph{attractive} Gaussian graphical model.
\end{itemize}
\end{block}

\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
In the Gaussian world, total positivity is a sign constraint on the inverse covariance matrix.
\end{beamercolorbox}
\end{frame}

\begin{frame}{Why this is less restrictive than it looks}
At first sight, $\mtp$ sounds severe.
But in sparse models or latent-factor models it often appears naturally.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Examples where $\mtp$\ arises}
\begin{itemize}
    \item one-factor models with positive loadings,
    \item latent tree models,
    \item ferromagnetic Ising models,
    \item Brownian motion tree models.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Economic intuition}
If a common hidden force pushes many variables in the same direction, positive dependence is natural.
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Statistical benefits}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 3: Why total positivity is statistically useful}}
\end{center}
\end{frame}

\begin{frame}{Why $\mtp$\ is useful for estimation}
The sign restriction is not just philosophically appealing.
It improves the statistics.

\begin{block}{Key facts in the Gaussian case}
\begin{itemize}
    \item the MLE under $\mtp$ exists with only two observations,
    \item the fitted inverse covariance is sparse,
    \item the constraint acts like a built-in regularizer.
\end{itemize}
\end{block}

\begin{alertblock}{Why economists should care}
This is highly relevant in finance and macro, where covariance estimation is hard and many variables are plausibly positively dependent.
\end{alertblock}
\end{frame}

\begin{frame}{A financial example}
Monthly correlations among major stock-market indices are often all positive.
A sample correlation matrix like
\[
S=\begin{pmatrix}
1 & 0.61 & 0.73\\
0.61 & 1 & 0.55\\
0.73 & 0.55 & 1
\end{pmatrix}
\]
already suggests a positive-dependence story.

\begin{block}{Why this matters}
If the fitted precision matrix also has non-positive off-diagonal entries,
then the Gaussian model is $\mtp$ and the whole dependence structure is being regularized by that sign information.
\end{block}
\end{frame}

\begin{frame}{A one-factor finance story again}
Suppose returns satisfy
\[
X_i = \lambda_i F + \varepsilon_i,
\qquad \lambda_i > 0,
\]
with a common market factor $F$.

\begin{alertblock}{Implication}
All cross-covariances are nonnegative:
\[
\cov(X_i,X_j)=\lambda_i\lambda_j\var(F)\geq 0.
\]
This is one reason positive dependence is natural in finance.
\end{alertblock}

\begin{block}{Message}
If your economic story is ``everything loads positively on the same latent factor'', then $\mtp$\ is not exotic at all.
\end{block}
\end{frame}

\section{Estimation under total positivity}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 4: Covariance estimation under total positivity}}
\end{center}
\end{frame}

\begin{frame}{The estimation problem}
Given a sample covariance matrix $S$, we want to estimate a covariance matrix under the constraint that the distribution is Gaussian $\mtp$.

\begin{block}{Equivalent formulation}
Estimate a positive definite precision matrix $K$ solving
\[
\min_{K\succ 0}
\Bigl\{-\log\det K + \operatorname{tr}(SK)\Bigr\}
\]
subject to
\[
K_{ij}\leq 0 \qquad (i\neq j).
\]
\end{block}

\begin{alertblock}{Interpretation}
This is a sign-constrained Gaussian graphical model.
The sign constraint replaces part of the usual tuning or regularization burden.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{A Gaussian $\mtp$\ fit in \texttt{R}}
A package route is to use \texttt{golazo} for sign-constrained Gaussian graphical models.

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
# if needed:
# devtools::install_github(
#   "pzwiernik/golazo")

library(golazo)
R <- cor(dat)
res <- positive.golazo(R, rho = Inf)
K <- res$K
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
library(qgraph)
qgraph(-cov2cor(K),
       layout = "spring",
       threshold = 0.05)
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
The estimated precision matrix respects the positive-dependence sign constraints and typically becomes sparse.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{A robust variation}
For heavy-tailed financial data, one can combine total positivity with rank-based correlations.

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
kR <- cor(dat, method = "kendall")
sR <- sin(pi * kR / 2)
res <- positive.golazo(sR, rho = Inf)
K <- res$K
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
qgraph(-cov2cor(K),
       layout = "spring")
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
This is the non-paranormal idea from Lecture 2 combined with the positive-dependence constraint.
\end{alertblock}
\end{frame}

\begin{frame}{What recent work says}
Recent work has pushed this in two directions:
\begin{itemize}
    \item more efficient algorithms for large Gaussian $\mtp$ graphical models,
    \item applications to covariance estimation and portfolio selection.
\end{itemize}

\begin{block}{Examples}
\begin{itemize}
    \item adaptive estimation of Gaussian graphical models under total positivity,
    \item covariance estimation under total positivity for portfolio selection,
    \item local and extremal variants of positive dependence beyond the classical Gaussian setting.
\end{itemize}
\end{block}
\end{frame}

\section{Binary and latent examples}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 5: Binary and latent-variable examples}}
\end{center}
\end{frame}

\begin{frame}{A binary latent-class example}
Positive dependence also appears naturally in binary data.

\begin{block}{Example}
Suppose three symptoms are conditionally independent given a latent binary disease state $H$:
\[
X_1 \indep X_2 \indep X_3 \mid H.
\]
If the disease state raises the probability of all three symptoms, then the observed distribution is positively dependent --- in fact often $\mtp$.
\end{block}

\begin{alertblock}{Interpretation}
A latent class model with a common “severity” variable often produces positive dependence among all observed symptoms.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Binary $\mtp$\ modeling in \texttt{R}}

\section{Conclusion}

\begin{frame}{Take-away}
\begin{itemize}
    \item Positive dependence is a real structural assumption, not just a descriptive comment.
    \item $\mtp$\ implies strong closure, monotonicity, and graphical properties.
    \item In Gaussian models, $\mtp$\ is equivalent to an M-matrix precision.
    \item This sign structure can be exploited for covariance and graph estimation.
    \item In finance and related applications, total positivity can be both economically plausible and statistically useful.
\end{itemize}

\vspace{2mm}
\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
Sparsity is not the only useful structure in dependence modeling.
The sign of dependence can be informative too.
\end{beamercolorbox}
\end{frame}
