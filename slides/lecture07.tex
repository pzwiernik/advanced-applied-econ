\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\huge \textcolor{DarkRed}{When treatment spills across a network}}
\end{center}
\end{frame}

\begin{frame}{What changes today?}
So far we treated each unit as if it lived in isolation.
Now we ask what happens when one unit's treatment affects another unit's outcome.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Examples economists care about}
\begin{itemize}
    \item peer effects in schools,
    \item information spillovers in villages,
    \item diffusion of incentives across firms,
    \item contagion in financial and production networks.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Main difficulty}
The treatment effect of unit $i$ may depend not only on $D_i$, but also on neighbors' treatments.
\end{block}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
\textbf{Main message:}
graphs do not only describe dependence --- they can also describe how treatment effects propagate.
\end{beamercolorbox}
\end{frame}

    \item potential outcomes and the role of SUTVA
    \item what breaks when SUTVA fails
    \item spillovers on graphs
    \item exposure mappings and simple estimands
    \item policy targeting under interference
\end{enumerate}

\section{Potential outcomes and SUTVA}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 1: Potential outcomes and SUTVA}}
\end{center}
\end{frame}

\begin{frame}{Quick review of potential outcomes}
For each unit $i$ and treatment level $d\in\{0,1\}$, define
\[
Y_i(d)
\]
as the outcome that would be observed under treatment status $d$.

\begin{block}{Average treatment effect}
\[
\tau = \E\{Y_i(1)-Y_i(0)\}.
\]
\end{block}

\begin{alertblock}{The classical picture}
Each unit has exactly two relevant potential outcomes:
one under treatment and one under control.
\end{alertblock}

\begin{block}{Interpretation}
This works well when units do not interfere with each other.
\end{block}
\end{frame}

\begin{frame}{SUTVA}
SUTVA stands for the \imp{Stable Unit Treatment Value Assumption}.

\begin{block}{Two parts}
\begin{itemize}
    \item \textbf{No multiple versions of treatment:} treatment $D_i=1$ means the same thing for all units.
    \item \textbf{No interference:} unit $i$'s outcome depends only on its own treatment, not on others' treatments.
\end{itemize}
\end{block}

\begin{alertblock}{Under SUTVA}
\[
Y_i(\mathbf d)=Y_i(d_i),
\]
so the potential outcome depends only on the unit's own treatment.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
SUTVA is what lets us write $Y_i(1)$ and $Y_i(0)$ without worrying about the rest of the population.
\end{beamercolorbox}
\end{frame}

\begin{frame}{Why SUTVA can fail}
Suppose students are connected by a friendship network.

\begin{center}
\begin{tikzpicture}[scale=0.9,
vertex/.style={circle,draw,very thick,minimum size=6mm,inner sep=0pt,fill=blue!10}]
\node[vertex] (1) at (0,0) {$1$};
\node[vertex] (2) at (1.6,0.8) {$2$};
\node[vertex] (3) at (3.2,0) {$3$};
\node[vertex] (4) at (1.6,-1.0) {$4$};
\draw[very thick] (1)--(2)--(3);
\draw[very thick] (2)--(4);
\draw[very thick] (1)--(4);
\end{tikzpicture}
\end{center}

\begin{block}{Example}
If unit $2$ receives tutoring, then unit $1$ may benefit indirectly through peer interaction.
\end{block}

\begin{alertblock}{Then}
\[
Y_1(\mathbf d)
\]
depends on the whole assignment vector $\mathbf d$, not only on $d_1$.
\end{alertblock}
\end{frame}

\begin{frame}{What changes when SUTVA fails?}
Without SUTVA, the potential outcome is indexed by the whole treatment assignment:
\[
Y_i(\mathbf d), \qquad \mathbf d=(d_1,\ldots,d_n).
\]

\begin{block}{Combinatorial explosion}
With $n$ units and binary treatment, there are
\[
2^n
\]
possible assignments, so each unit may have an enormous number of potential outcomes.
\end{block}

\begin{alertblock}{Main challenge}
We need structure to reduce this complexity.
Networks and exposure mappings are one way to do it.
\end{alertblock}
\end{frame}

\section{Spillovers on graphs}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 2: Spillovers on graphs}}
\end{center}
\end{frame}

\begin{frame}{A graph-based view of interference}
Let $G$ be a graph describing who can affect whom.

\begin{center}
\begin{tikzpicture}[scale=0.9,
vertex/.style={circle,draw,very thick,minimum size=6mm,inner sep=0pt,fill=blue!10}]
\node[vertex] (1) at (0,0) {$1$};
\node[vertex] (2) at (1.6,0.9) {$2$};
\node[vertex] (3) at (3.2,0) {$3$};
\node[vertex] (4) at (1.6,-1.0) {$4$};
\node[vertex] (5) at (4.8,0.9) {$5$};
\node[vertex] (6) at (4.8,-0.9) {$6$};
\draw[very thick] (1)--(2)--(3)--(5);
\draw[very thick] (2)--(4)--(3);
\draw[very thick] (3)--(6);
\end{tikzpicture}
\end{center}

\begin{block}{Interpretation}
The graph restricts the set of possible spillovers:
unit $i$ may be affected by treatments assigned to its neighbors, but perhaps not by distant units.
\end{block}

\begin{alertblock}{This is the key modeling move}
The network is what makes interference manageable.
\end{alertblock}
\end{frame}

\begin{frame}{Exposure mappings}
A common simplification is to replace the full assignment vector by an exposure summary.

\begin{block}{Example}
Instead of writing $Y_i(\mathbf d)$, write
\[
Y_i(d_i, e_i),
\]
where $e_i$ is an exposure measure such as:
\begin{itemize}
    \item number of treated neighbors,
    \item share of treated neighbors,
    \item whether at least one neighbor is treated.
\end{itemize}
\end{block}

\begin{alertblock}{Interpretation}
Two treatment assignments that produce the same $(d_i,e_i)$ are assumed to generate the same potential outcome for unit $i$.
\end{alertblock}
\end{frame}

\begin{frame}{Direct and spillover effects}
If exposure is summarized by $e_i$, we can define several effects.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Direct effect}
Compare
\[
Y_i(1,e)-Y_i(0,e),
\]
holding neighbor exposure fixed.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Spillover effect}
Compare
\[
Y_i(d,e')-Y_i(d,e),
\]
holding own treatment fixed.
\end{block}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
Under interference, “the treatment effect” splits into a direct component and a spillover component.
\end{beamercolorbox}
\end{frame}

\begin{frame}{A village example}
Suppose treatment is access to information about fertilizer.

\begin{itemize}
    \item $D_i=1$ means household $i$ receives the information directly.
    \item Neighbors who receive the information may also affect household $i$ through discussion or imitation.
\end{itemize}

\begin{alertblock}{Potential outcomes}
A natural model is
\[
Y_i(d_i, e_i),
\]
where $e_i$ is the proportion of treated neighbors.
\end{alertblock}

\begin{block}{Interpretation}
\begin{itemize}
    \item direct effect: effect of being treated yourself,
    \item spillover effect: effect of living in a more treated neighborhood.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{A school or peer-effects example}
Suppose treatment is assignment to a tutoring program.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Own-treatment effect}
Does the program improve the treated student's outcome?
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Peer spillover}
Does having more treated friends improve the untreated student's outcome?
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Why this matters}
A policy may produce benefits even for untreated units.
If we ignore that, we may badly mis-measure policy impact.
\end{alertblock}
\end{frame}

\section{A simple simulation}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 3: A tiny simulation}}
\end{center}
\end{frame}

\begin{frame}[fragile]{A toy spillover simulation in \texttt{R}}
Suppose outcomes satisfy
\[
Y_i = 1.0\,D_i + 0.8\,E_i + \varepsilon_i,
\]
where $E_i$ is the share of treated neighbors.

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
set.seed(1)
n <- 50
A <- matrix(0, n, n)
for (i in 1:(n-1)) {
  A[i, i+1] <- 1
  A[i+1, i] <- 1
}

D <- rbinom(n, 1, 0.4)
deg <- rowSums(A)
E <- (A %*% D) / pmax(deg, 1)
Y <- 1.0 * D + 0.8 * E + rnorm(n)
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
summary(lm(Y ~ D))
summary(lm(Y ~ D + E))
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
If we regress only on $D$, the spillover term is omitted.
Including an exposure measure $E$ lets us separate the direct and spillover components.
\end{alertblock}
\end{frame}

\begin{frame}{What this simulation shows}
The graph does not only generate dependence.
It also determines how treatment reaches other units.

\begin{block}{Lesson}
If spillovers are real, then an analysis that ignores exposure may:
\begin{itemize}
    \item overstate or understate the direct effect,
    \item miss important social multipliers,
    \item and suggest the wrong policy ranking.
\end{itemize}
\end{block}

\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
Interference turns a treatment assignment problem into a network assignment problem.
\end{beamercolorbox}
\end{frame}

\section{Targeting and policy evaluation}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 4: Targeting and policy evaluation under interference}}
\end{center}
\end{frame}

\begin{frame}{Why targeting is harder under interference}
Without interference, a planner can ask:
\begin{center}
Who benefits most from treatment?
\end{center}

With interference, the planner must instead ask:
\begin{center}
Which pattern of assignments creates the most social value on the network?
\end{center}

\begin{alertblock}{Main difference}
Treating one unit may help or hurt others.
So the best policy depends on the whole assignment, not only on unit-by-unit treatment effects.
\end{alertblock}
\end{frame}

\begin{frame}{Davide Viviano as a modern anchor}
A nice modern entry point is Davide Viviano's work on policy targeting under network interference.

\begin{block}{Main idea}
Instead of treating units independently, choose treatment assignments to maximize average welfare when spillovers are present.
\end{block}

\begin{alertblock}{Why this is attractive for applied economics}
\begin{itemize}
    \item treatment effects may be heterogeneous,
    \item networks may matter for welfare,
    \item policy design becomes an optimization problem under interference.
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{A very simple targeting picture}
Suppose a treatment has both direct and spillover benefits.
Then a planner may prefer to treat
\begin{itemize}
    \item central nodes,
    \item bridges between groups,
    \item or units whose treatment benefits many untreated neighbors.
\end{itemize}

\begin{block}{Contrast with the no-interference case}
Without spillovers, targeting is mostly about the treated unit.
With spillovers, targeting is also about the unit's position in the network.
\end{block}
\end{frame}

\begin{frame}{Packages and practical tools}
There is no single dominant package for all interference problems, but there are useful tools.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{R}
\begin{itemize}
    \item \texttt{RCT2}: two-stage randomized designs with interference,
    \item \texttt{latenetwork}: IV-style estimation under network interference,
    \item \texttt{igraph}: graph construction and network summaries.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Python}
For general graph handling, \texttt{networkx} is useful.
For causal spillover estimation, practice is more fragmented and often custom.
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Practical message}
For teaching and small projects, it is often best to combine a simple exposure mapping with standard regression or randomization-based logic.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{A package-style example in \texttt{R}}
For two-stage randomized designs with interference, \texttt{RCT2} and \texttt{latenetwork} provide useful starting points.

\begin{columns}[T,onlytextwidth]

\section{Conclusion}

\begin{frame}{Take-away}
\begin{itemize}
    \item Under SUTVA, each unit's potential outcome depends only on its own treatment.
    \item Under network interference, outcomes may depend on the full assignment vector.
    \item Exposure mappings use graph structure to simplify the problem.
    \item Direct and spillover effects are conceptually distinct.
    \item Policy targeting under interference is about assignments on a graph, not isolated units.
\end{itemize}

\vspace{2mm}
\begin{beamercolorbox}[wd=\textwidth,sep=8pt,center,rounded=true,shadow=true]{important}
Interference is one of the clearest examples of how graphs enter applied economics not only descriptively, but directly in identification and policy design.
\end{beamercolorbox}
\end{frame}
