\section{Directed acyclic graphs}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 1: DAGs as structural models}}
\end{center}
\end{frame}

\begin{frame}{What is a DAG?}
A \imp{directed acyclic graph} (DAG) consists of
\begin{itemize}
    \item vertices $1,\ldots,m$ representing variables $X_1,\ldots,X_m$,
    \item directed edges $i\to j$,
    \item no directed cycles.
\end{itemize}
\vspace{-4mm}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=0.8,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (1) at (0,0) {$1$};
\node[vertex] (2) at (2,1.2) {$2$};
\node[vertex] (3) at (2,-1.2) {$3$};
\node[vertex] (4) at (4,0) {$4$};
\draw[->,very thick] (1)--(2);
\draw[->,very thick] (1)--(3);
\draw[->,very thick] (2)--(4);
\draw[->,very thick] (3)--(4);
\end{tikzpicture}
\end{center}
\vspace{-4mm}
\begin{alertblock}{Interpretation}
The arrow $i\to j$ says that $X_i$ is a direct input into the mechanism generating $X_j$.
\end{alertblock}
\end{frame}

\begin{frame}{Parents, children, ancestors, descendants}
Let $D$ be a DAG.

\begin{block}{}
For a node $j$:\\[2mm]
\begin{minipage}{7cm}
	\begin{itemize}
    \item $\pa(j)$ = parents of $j$,
    \item $\ch(j)$ = children of $j$,
\end{itemize}
\end{minipage}\begin{minipage}{7cm}
	\begin{itemize}
    \item $\an(j)$ = ancestors of $j$,
    \item $\de(j)$ = descendants of $j$.
\end{itemize}
\end{minipage}
\end{block}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=0.95,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (1) at (0,1.2) {$1$};
\node[vertex] (2) at (0,0) {$2$};
\node[vertex] (3) at (2,0) {$3$};
\node[vertex] (4) at (4,0) {$4$};
\node[vertex] (5) at (4,1.2) {$5$};
\draw[->,very thick] (1)--(2);
\draw[->,very thick] (2)--(3);
\draw[->,very thick] (3)--(4);
\draw[->,very thick] (1)--(5);
\draw[->,very thick] (5)--(4);
\end{tikzpicture}
\end{center}
\end{column}
\begin{column}{0.46\textwidth}
For node $4$:
\[
\pa(4)=\{3,5\},
\qquad
\an(4)=\{1,2,3,5\}.
\]
For node $1$:
\[
\ch(1)=\{2,5\},
\qquad
\de(1)=\{2,3,4,5\}.
\]
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Factorization over a DAG}
A distribution $P$ is \imp{Markov with respect to a DAG $D$} if its density factorizes as
\[
p(x_1,\ldots,x_m)=\prod_{j=1}^m p\bigl(x_j\mid x_{\pa(j)}\bigr).
\]

\begin{block}{Interpretation}
Each variable is generated from its parents and no other variables enter directly.
\end{block}

\begin{alertblock}{Why this matters}
The graph now has a dual role:
\begin{itemize}
    \item it determines a factorization of the joint law,
    \item it implies conditional independence statements.
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{A simple factorization example}
Consider the DAG

\begin{center}
\begin{tikzpicture}[scale=.7,
    >=Stealth,
    node distance=2.1cm,
    every node/.style={
        circle,
        draw,
        very thick,
        minimum size=7mm,
        inner sep=0pt,
        font=\bfseries
    }
]
\node (1) at (0,0) {$1$};
\node (2) at (2,1.2) {$2$};
\node (4) at (2,-1.2) {$4$};
\node (3) at (4,0) {$3$};

\draw[->,very thick] (1) -- (2);
\draw[->,very thick] (1) -- (4);
\draw[->,very thick] (2) -- (3);
\draw[->,very thick] (4) -- (3);
\end{tikzpicture}
\end{center}\vspace{-4mm}
Then the joint law factorizes as
\[
p(x_1,x_2,x_3,x_4)
=
p(x_1)\,p(x_2\mid x_1)\,p(x_4\mid x_1)\,p(x_3\mid x_2,x_4).
\]
\vspace{-7mm}
\begin{block}{}
Think of $X_1$ as an initial state, $X_2$ and $X_4$ as intermediate channels,
and $X_3$ as the final outcome.
\end{block}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
In a DAG, each factor involves one variable and its parents.
\end{beamercolorbox}
\end{frame}

\begin{frame}{From DAGs to structural equations}
A DAG can be viewed not only as a factorization of a joint law, but also as a system of structural assignments.

\begin{block}{Structural equation view}
For each node $j$, write
\[
X_j = f_j\bigl(X_{\pa(j)},\varepsilon_j\bigr),
\qquad j=1,\ldots,m,
\]
where the noise variables $\varepsilon_1,\ldots,\varepsilon_m$ are jointly independent.
\end{block}

\begin{alertblock}{Why this matches the DAG}
Each variable depends only on its parents and on its own noise term.
So the arrows describe which variables enter directly into each mechanism.
\end{alertblock}
\end{frame}

\begin{frame}{Local Markov property}
For each node $j$ in a DAG,
\[
X_j \indep X_{\nd(j)\setminus \pa(j)}\mid X_{\pa(j)},
\]
where $\nd(j)$ denotes the non-descendants of $j$.
\medskip

\begin{block}{Interpretation}
Once we know the parents of $X_j$, all non-descendants of $j$ become irrelevant for predicting $X_j$.
\end{block}

\begin{alertblock}{Why this is natural}
This says that all direct inputs into $X_j$ have already been listed as parents.
\end{alertblock}
\end{frame}

\begin{frame}{The chain pattern}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=1.0,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (x) at (0,0) {$X$};
\node[vertex,fill=red!20] (z) at (2,0) {$Z$};
\node[vertex] (y) at (4,0) {$Y$};
\draw[->,very thick] (x)--(z)--(y);
\end{tikzpicture}
\end{center}

\begin{alertblock}{Interpretation}
$Z$ is a transmission channel between $X$ and $Y$.
Once we condition on $Z$, the path is blocked:
\[
X\indep Y\mid Z.
\]
\end{alertblock}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Policy example}
Policy $X$ changes investment $Z$, which changes output $Y$.
After fixing investment, policy adds no further predictive content for output.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Finance example}
Monetary news $X$ moves interest rates $Z$, which then affect stock returns $Y$.
After fixing interest rates, monetary news adds no further predictive content for returns.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{The fork pattern}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=1.0,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (x) at (0,0) {$X$};
\node[vertex,fill=red!20] (z) at (2,0) {$Z$};
\node[vertex] (y) at (4,0) {$Y$};
\draw[->,very thick] (z)--(x);
\draw[->,very thick] (z)--(y);
\end{tikzpicture}
\end{center}

\begin{alertblock}{Interpretation}
$Z$ is a common cause of $X$ and $Y$.
Once we condition on $Z$, the path is blocked:
\[
X\indep Y\mid Z.
\]
\end{alertblock}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Demand-shock example}
A common demand shock $Z$ affects both price $X$ and quantity $Y$.
After controlling for the shock, the induced association disappears.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Ability example}
Latent ability $Z$ affects both education $X$ and earnings $Y$.
Conditioning on ability removes the spurious component of the association.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{The collider pattern}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=1.0,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (x) at (0,0) {$X$};
\node[vertex,fill=red!20] (z) at (2,0) {$Z$};
\node[vertex] (y) at (4,0) {$Y$};
\draw[->,very thick] (x)--(z);
\draw[->,very thick] (y)--(z);
\end{tikzpicture}
\end{center}
\vspace{-4mm}
\begin{alertblock}{Interpretation}
Without conditioning, the path is blocked at the collider, so typically $X\indep Y$.\\[3mm]

Conditioning on $Z$ opens the path and can create dependence (we discussed this in Week 1).
\end{alertblock}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Selection example}
Admission $Z$ to a school or program may depend on both test scores $X$ and extracurricular achievements $Y$.
Among admitted students, these two variables may appear negatively related.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Labor-market example}
Staying in a demanding occupation $Z$ may depend on both ability $X$ and motivation $Y$.
Conditioning on those who remain in the occupation can induce an artificial dependence between them.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{d-separation: the general rule}
A path is \imp{active given $S$} if
\begin{itemize}
    \item every non-collider on the path is outside $S$,
    \item every collider on the path has itself or one of its descendants in $S$.
\end{itemize}

Equivalently, a path is \imp{blocked given $S$} if it is not active given $S$.

\begin{alertblock}{d-separation}
Two sets  $A$ and $B$ are \imp{d-separated by $S$}
if every path between $A$ and $B$ is blocked given $S$.
\end{alertblock}

\begin{block}{Markov property for DAGs}
If $A$ and $B$ are d-separated by $S$ in the DAG, then
\[
X_A\indep X_B\mid X_S.
\]
\end{block}
\end{frame}

\begin{frame}{A longer example}
Consider the DAG
\vspace{-6mm}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=.8,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (1) at (0,0) {$1$};
\node[vertex] (2) at (1.7,0) {$2$};
\node[vertex] (4) at (3.5,0) {$4$};
\node[vertex] (3) at (3.5,-1.5) {$3$};
\node[vertex] (5) at (1.7,1.5) {$5$};
\draw[->,very thick] (1)--(2);
\draw[->,very thick] (2)--(4);
\draw[->,very thick] (3)--(4);
\draw[->,very thick] (2)--(5);
\end{tikzpicture}
\end{center}

\begin{itemize}
    \item $1$ and $3$ are d-separated by $\emptyset$ because the only path hits the collider $4$.
    \item $1$ and $3$ become d-connected if we condition on $4$.
    \item $1$ and $4$ are d-separated by $\{2\}$ because the chain is blocked at $2$.
\end{itemize}
\end{frame}

\begin{frame}{Moralization: a practical way to test d-separation}
A useful way to test d-separation is to convert the DAG into an undirected graph.

\begin{enumerate}
    \item Keep only the ancestors of $A\cup B\cup S$.
    \item \textbf{Moralize}: connect any two parents of a common child.
    \item Drop all arrow directions.
    \item Check whether $S$ separates $A$ and $B$ in the resulting undirected graph.
\end{enumerate}

\begin{alertblock}{Moralization criterion}
$A$ and $B$ are d-separated by $S$ in the DAG
if and only if $S$ separates $A$ and $B$ in the moralized ancestral graph.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
This reduces a d-separation question in a DAG to an ordinary separation question in an undirected graph.
\end{beamercolorbox}
\end{frame}

\section{Markov equivalence}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 3: Markov equivalence and CPDAGs}}
\end{center}
\end{frame}

\begin{frame}{Markov equivalence: why observational data do not identify every arrow}
Different DAGs can encode the same set of conditional independence relations.

\begin{block}{Definition}
Two DAGs are \imp{Markov equivalent} if they imply the same conditional independence statements.
\end{block}

\begin{center}
\begin{tikzpicture}[
>=Latex,
scale=0.95,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}
]
% first DAG
\node[vertex] (a1) at (0,0) {$1$};
\node[vertex] (a2) at (1.4,0) {$2$};
\node[vertex] (a3) at (2.8,0) {$3$};
\draw[->,very thick] (a1)--(a2);
\draw[->,very thick] (a2)--(a3);

% second DAG
\node[vertex] (b1) at (4.6,0) {$1$};
\node[vertex] (b2) at (6.0,0) {$2$};
\node[vertex] (b3) at (7.4,0) {$3$};
\draw[->,very thick] (b2)--(b1);
\draw[->,very thick] (b3)--(b2);

% third DAG
\node[vertex] (c1) at (9.2,0) {$1$};
\node[vertex] (c2) at (10.6,0) {$2$};
\node[vertex] (c3) at (12.0,0) {$3$};
\draw[->,very thick] (c2)--(c1);
\draw[->,very thick] (c2)--(c3);
\end{tikzpicture}
\end{center}
\vspace{-4mm}
\begin{alertblock}{Equivalent example}
These three DAGs are Markov equivalent.
All of them imply $X_1\indep X_3\mid X_2$.
\end{alertblock}
\begin{alertblock}{Non-equivalent collider}
\begin{tikzpicture}[
>=Latex,
scale=0.95,
vertex/.style={circle,draw,very thick,minimum size=5mm,inner sep=0pt}
]
\node[vertex] (d1) at (0,0) {$1$};
\node[vertex] (d2) at (1.4,0) {$2$};
\node[vertex] (d3) at (2.8,0) {$3$};
\draw[->,very thick] (d1)--(d2);
\draw[->,very thick] (d3)--(d2);
\end{tikzpicture} is \textbf{not} equivalent to the three above, because it implies
$X_1\indep X_3$.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
From observational data, we can typically recover only a Markov equivalence class.
\end{beamercolorbox}
\end{frame}
\begin{frame}{The Verma--Pearl characterization}
Two DAGs are Markov equivalent if and only if they have
\begin{itemize}
    \item the same \imp{skeleton} (same undirected graph),
    \item the same \imp{v-structures}.
\end{itemize}

\begin{block}{v-structure}
A triple of the form
\[
X\to Z\leftarrow Y
\]
with no edge between $X$ and $Y$.
\end{block}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Observational data identify the skeleton and the colliders, but often not all arrow directions.
\end{beamercolorbox}
\end{frame}

\begin{frame}{CPDAGs / essential graphs}
A \imp{CPDAG} represents an entire Markov equivalence class.

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\begin{block}{}
\begin{itemize}
    \item \textbf{Directed edges:} directions shared by \emph{every} DAG in the class.
    \item \textbf{Undirected edges:} directions that cannot be identified from observational data alone.
\end{itemize}
\end{block}

\begin{alertblock}{Important point}
All v-structures are fixed across the equivalence class,
but there may be more arrows. 
\end{alertblock}
\end{column}

\begin{column}{0.46\textwidth}
\begin{center}
\begin{tikzpicture}[
>=Latex,
scale=.6,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}
]
\node[vertex] (1) at (0,1) {$1$};
\node[vertex] (2) at (3,0) {$2$};
\node[vertex] (3) at (0,-1) {$3$};
\node[vertex] (4) at (6.5,0) {$4$};

\draw[->,very thick] (1)--(2);
\draw[->,very thick] (3)--(2);
\draw[->,very thick] (2)--(4);
\end{tikzpicture}
\end{center}

\vspace{1mm}

\begin{block}{Example}
Above $1\to2\leftarrow 3$ is a v-structure but the arrow $2\to4$ is also determined.\end{block}
\end{column}
\end{columns}

\vspace{1mm}
\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}\small 
A CPDAG separates what is identifiable from observational data (with no extra assumptions) from what remains unresolved within the equivalence class.
\end{beamercolorbox}
\end{frame}

\section{Interventions}

\begin{frame}[plain,noframenumbering]{}
\begin{center}

\end{center}
\end{frame}

\begin{frame}{Observing is not the same as intervening}
Observational data tell us how variables move together. Policy analysis asks a different question:
\begin{center}
\large \textcolor{DarkBlue}{What happens if we change one of the mechanisms?}
\end{center}
\vspace{-3mm}
\begin{columns}[T]
\begin{column}{0.38\textwidth}
\begin{center}
\begin{tikzpicture}[>=Latex,scale=0.75,
vertex/.style={circle,draw,very thick,minimum size=5mm,inner sep=0pt}]
\node[vertex] (u) at (0,0) {$U$};
\node[vertex] (x) at (2,0.9) {$X$};
\node[vertex] (y) at (4,0) {$Y$};
\draw[->,very thick] (u)--(x);
\draw[->,very thick] (u)--(y);
\draw[->,very thick] (x)--(y);
\end{tikzpicture}
\end{center}
\vspace{-4mm}
\begin{block}{Example}
$X$ = policy variable\\
$Y$ = economic outcome\\
$U$ = background conditions
\end{block}
\end{column}
\begin{column}{0.57\textwidth}
\begin{block}{\normalsize \textbf{Observational question:} What is \alert{$\P(Y\mid X=x)$}?}
Outcomes among units with observed value $X=x$.
\end{block}
\vspace{-3mm}
\begin{block}{\normalsize \textbf{Interventional question:} What is
\alert{$
\P(Y\mid \operatorname{do}(X=x))$}?
}
This describes outcomes if we actively set $X$ to $x$.
\end{block}
If $U$ affects both $X$ and $Y$, then\\[2mm]
\qquad\qquad$
\P(Y\mid X=x)\;\neq\; \P(Y\mid \operatorname{do}(X=x)).
$
\end{column}
\end{columns}
\smallskip
\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
\small Association describes the world as it is; intervention asks what happens if we change the system.
\end{beamercolorbox}
\end{frame}

\begin{frame}{Hard interventions: the \texorpdfstring{$do$}{do} operator}
A hard intervention sets one variable externally:\qquad
$
\operatorname{do}(X_k=x).
$

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{center}
\textbf{Observational regime}

\begin{tikzpicture}[>=Latex,scale=0.95,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (u) at (0,0) {$U$};
\node[vertex] (x) at (2,0.9) {$X$};
\node[vertex] (y) at (4,0) {$Y$};
\draw[->,very thick] (u)--(x);
\draw[->,very thick] (u)--(y);
\draw[->,very thick] (x)--(y);
\end{tikzpicture}
\end{center}
\end{column}

\begin{column}{0.48\textwidth}
\begin{center}
\textbf{After $\operatorname{do}(X=x)$}

\begin{tikzpicture}[>=Latex,scale=0.95,
vertex/.style={circle,draw,very thick,minimum size=7mm,inner sep=0pt}]
\node[vertex] (u) at (0,0) {$U$};
\node[vertex,fill=red!20] (x) at (2,0.9) {$X$};
\node[vertex] (y) at (4,0) {$Y$};
\draw[->,very thick] (u)--(y);
\draw[->,very thick] (x)--(y);
\end{tikzpicture}
\end{center}
\end{column}
\end{columns}

\begin{alertblock}{Graphical meaning}
A hard intervention removes all arrows pointing into the intervened variable.
So the mechanism generating $X$ is replaced by an external assignment.
\end{alertblock}

\begin{block}{Economic reading}
This is the idealized version of a randomized policy assignment:
the policy variable is no longer chosen in response to background conditions.
\end{block}
\end{frame}

\begin{frame}{Soft interventions: changing a mechanism}
In economics, policy often changes a mechanism without fixing a variable to a constant.

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Observational SEM}
$
X = g(U,\varepsilon_X),
\qquad
Y = h(X,U,\varepsilon_Y).
$
\end{block}

\begin{block}{Soft intervention}
A policy may change the equation for $Y$ into\\
\qquad \qquad $
Y = h^\star(X,U,\varepsilon_Y),
$\\
or change parameters inside $h$.
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{exampleblock}{Examples}
\begin{itemize}
    \item a tax changes incentives,
    \item a subsidy shifts demand,
    \item monetary policy changes a reaction function,
    \item a regulation changes firms' decision rules.
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}

\begin{alertblock}{Key idea}
A soft intervention changes one structural mechanism while leaving most of the system intact.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Policy analysis is often about \textbf{mechanism shifts}, not just fixing variables by force.
\end{beamercolorbox}
\end{frame}



\begin{frame}{How soft interventions can reveal causal direction}
Suppose the true causal direction is
$
X \longrightarrow Y,
$
with structural equations
\[
X = f(\varepsilon_X),
\qquad
Y = g(X,\varepsilon_Y).
\]

Now imagine a soft intervention that changes only the mechanism for $X$:
\[
X = f^\star(\varepsilon_X),
\qquad
Y = g(X,\varepsilon_Y).
\]
\vspace{-7mm}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{block}{What changes?}
Because the mechanism for $X$ changes, generally\\[1mm]
\qquad \qquad $
P^\star(X)\neq P(X),
\qquad
P^\star(Y)\neq P(Y),
$\\[1mm]
and typically also\\[1mm]
\qquad \qquad \qquad $
P^\star(X\mid Y)\neq P(X\mid Y).
$
\end{block}
\end{column}

\begin{column}{0.42\textwidth}
\begin{block}{What stays the same?}
The mechanism from $X$ to $Y$ is unchanged, so
\[
P^\star(Y\mid X)=P(Y\mid X).
\]
\end{block}
\end{column}
\end{columns}

\smallskip
\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
This is only an intuition, but it is the basic idea behind several methods based on invariance across environments.
\end{beamercolorbox}
\end{frame}




\begin{frame}[fragile]{A tiny simulation: invariant mechanism, changing input distribution}
Suppose the true direction is
\[
X \to Y,
\qquad
Y = X^2 + \varepsilon_Y,
\]
and only the distribution of $X$ changes across environments.

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{lstlisting}
set.seed(1)
n <- 1000

x0 <- rnorm(n, 0, 1)
y0 <- x0^2 + rnorm(n, sd = 0.5)

x1 <- rnorm(n, 2, 1)
y1 <- x1^2 + rnorm(n, sd = 0.5)
\end{lstlisting}
\end{column}
\begin{column}{0.48\textwidth}
\begin{lstlisting}
par(mfrow = c(1,2))
hist(x0, freq = FALSE, main = "X")
hist(x1, freq = FALSE, add = TRUE)

plot(x0, y0, pch = 16, col = 1)
points(x1, y1, pch = 16, col = 2)
\end{lstlisting}
\end{column}
\end{columns}

\begin{alertblock}{Interpretation}
The marginals change across environments, but the conditional mechanism relating $Y$ to $X$ remains the same.
\end{alertblock}
\end{frame}

\begin{frame}{Why histograms and conditional fits can reveal direction}
In the simulation, the intervention changed the distribution of the cause $X$.

\begin{block}{Across environments}
\begin{itemize}
    \item the histograms of $X$ and of $Y$ change,
    \item but the conditional relation $Y\mid X$ stays stable.
\end{itemize}
\end{block}

\begin{alertblock}{Wrong reverse direction}
If we tried to model the system as
$
Y \to X$, we would need the mechanism $X\mid Y$ to remain stable across environments.
In general this fails.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
To identify direction, look for the orientation in which the conditional mechanism remains stable when environments change.
\end{beamercolorbox}
\end{frame}

\section{Linear SEMs}

\begin{frame}[plain,noframenumbering]{}
\begin{center}
{\Huge \textcolor{DarkRed}{Part 5: Linear SEMs and path tracing}}
\end{center}
\end{frame}

\begin{frame}{Linear structural equation models}
A linear SEM associated with a DAG has the form
\[
X_i = \sum_{j\in \pa(i)} \beta_{ij} X_j + \varepsilon_i,
\qquad i=1,\ldots,m,
\]
where $\varepsilon_1,\ldots,\varepsilon_m$ are mutually independent
(or at least satisfy $\E(\varepsilon_i\mid X_{\pa(i)})=0$).

\begin{columns}[T]
\begin{column}{0.62\textwidth}
\begin{block}{}
\textbf{Matrix form:}
$
X = BX + \varepsilon,
$
where
$
B_{ij}=0 $ when  $j\notin \pa(i)$.
\end{block}\vspace{-2mm}

\begin{alertblock}{Note}
The zero pattern of $B$ is exactly the DAG.
\end{alertblock}\vspace{9mm}

\begin{alertblock}{Interpretation of the example}
$X_1$ is a primitive shock, $X_2$ and $X_3$ are intermediate channels,
and $X_4$ is the outcome of interest.
\end{alertblock}
\end{column}

\begin{column}{0.35\textwidth}
\begin{beamercolorbox}[wd=\textwidth,sep=0pt]{examplepanel}
\[
\begin{aligned}
X_1 &= \varepsilon_1,\\
X_2 &= \beta_{21}X_1+\varepsilon_2,\\
X_3 &= \beta_{31}X_1+\varepsilon_3,\\
X_4 &= \beta_{42}X_2+\beta_{43}X_3+\varepsilon_4.
\end{aligned}
\]

\vspace{-6mm}

\begin{frame}{Gaussian LSEMs are exactly Gaussian DAG models}

{\small Recall: A DAG model is defined by the factorization
$
f(x)=\prod_{i=1}^m f(x_i\mid x_{\pa(i)}).
$}

\begin{block}{Gaussian DAG model}
If $X$ is jointly Gaussian, then each conditional distribution
$
X_i \mid X_{\pa(i)}
$
is Gaussian.
So a Gaussian DAG model is specified by choosing these Gaussian conditionals.
\end{block}

\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{block}{}
Gaussian LSEM writes\\[1mm]
\qquad\qquad$
X_i=\sum_{j\in\pa(i)} \beta_{ij}X_j+\varepsilon_i$,\quad 
$\varepsilon_i\sim N(0,\omega_i)$,\\[1mm] where the errors $\varepsilon_i$ are independent Gaussian.
\end{block}
\end{column}

\begin{column}{0.42\textwidth}
\begin{alertblock}{Conditional law}
\ \\
\quad $
X_i\mid X_{\pa(i)}
\sim
N\!\left(
\sum_{j\in\pa(i)} \beta_{ij}X_j,
\omega_i
\right).
$
\end{alertblock}
\end{column}
\end{columns}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
So a linear Gaussian SEM specifies exactly the Gaussian conditionals appearing in the DAG factorization.
\end{beamercolorbox}
\end{frame}



\begin{frame}{Structural coefficients and total effects}

\begin{block}{Structural coefficients}
In the linear SEM
\[
X = BX + \varepsilon,
\]
the matrix $B$ records the \imp{direct} linear effects encoded by the DAG.
\end{block}

\begin{block}{Reduced-form map}
Solving the system gives
\[
X = (I-B)^{-1}\varepsilon.
\]
So the matrix $(I-B)^{-1}$ combines \imp{direct and indirect} effects.
\end{block}

%\begin{alertblock}{Interpretation}
%The DAG tells us which effects are direct.
%The matrix $(I-B)^{-1}$ tells us how shocks propagate through the whole system.
%\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Structural coefficients describe local mechanisms.  
The reduced-form map describes the total propagation of shocks.
\end{beamercolorbox}
\end{frame}

\begin{frame}{Wright's path tracing: the basic idea}
In a \textbf{linear} SEM, the covariance between two variables can be written as contributions from different paths in the graph.

\begin{columns}[T]
\begin{column}{0.56\textwidth}

\begin{block}{}
\textbf{Why this is useful: }It helps explain an observed association as a combination of
\begin{itemize}
    \item direct effects,
    \item indirect effects through mediators,
    \item common-cause effects.
\end{itemize}
\end{block}
\vspace{-3mm}
\begin{alertblock}{Path tracing calculation}
To understand $\cov(X_1,X_4)$, we look at the two directed paths
$
1\to 2\to 4$, $1\to 3\to 4$.
Each path contributes the product of its coefficients, so\\[2mm]
\qquad\qquad$
\cov(X_1,X_4)
=
\beta_{21}\beta_{42}
+
\beta_{31}\beta_{43}.
$
\end{alertblock}
\end{column}
\begin{column}{0.33\textwidth}
\begin{beamercolorbox}[wd=\textwidth,sep=0pt]{examplepanel}
\[
\begin{aligned}
X_2 &= \beta_{21}X_1+\varepsilon_2,\\
X_3 &= \beta_{31}X_1+\varepsilon_3,\\
X_4 &= \beta_{42}X_2+\beta_{43}X_3+\varepsilon_4,
\end{aligned}
\]
with $\var(X_1)=1$.
\vspace{1mm}

\begin{frame}{A caution about interpretation}
DAGs and SEMs are powerful, but they do not identify themselves from thin air.

\begin{itemize}
    \item d-separation gives testable implications,
    \item Markov equivalence limits what observational data can identify,
    \item structural interpretation always rests on substantive assumptions.
\end{itemize}

\begin{alertblock}{Important caution}
A DAG is not automatically discovered just because we fit a model.
It is a structured hypothesis about the data-generating process.
\end{alertblock}

\begin{beamercolorbox}[wd=\textwidth,sep=7pt,center,rounded=true,shadow=true]{important}
Graphs are useful because they make assumptions explicit, not because they eliminate the need for assumptions.
\end{beamercolorbox}
\end{frame}
